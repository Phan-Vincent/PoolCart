# PoolCart — Grocery Group Buy Mobile App (Expo + React Native)

PoolCart is a cross-platform mobile application (iOS, Android, and Web via Expo) built for **pooling neighborhood grocery orders to hit wholesale/delivery minimums** AND **splitting bulk cases** (like a 6-pack of barista oat milk or a 50 lb sack of jasmine rice) across neighbors.

## Why Start With This Architecture?

1. **Order Pooling + Case Splitting Hybrid**:
   - **Direct Wholesale Items (`individual` mode)**: Neighbors order full units (e.g., a 12-count bag of avocados or a 30-egg flat) that combine toward the supplier's **$250 minimum order** and **$300 free delivery tier**.
   - **Bulk Case Splits (`case_split` mode)**: Neighbors claim individual shares of a bulk wholesale case (e.g., 1 carton out of a 6-pack case). Visual slot bars track how many shares are claimed (`4/6 claimed`) and automatically roll completed cases into the pool total.
2. **Pragmatic Backend Path (Local-First $\rightarrow$ Supabase)**:
   - **Right now (Zero Setup)**: Uses `AsyncStorage` (`src/services/repository.ts`) so you can run the app immediately, test adding items, claim shares, and mark neighbors as paid without configuring cloud keys.
   - **When inviting real neighbors**: Run `supabase/schema.sql` in a free [Supabase](https://supabase.com/) project and add `EXPO_PUBLIC_SUPABASE_URL` and `EXPO_PUBLIC_SUPABASE_ANON_KEY` to a `.env` file. Supabase gives you Postgres relational integrity (preventing two people from claiming the last share of a case at the same time) plus **Supabase Realtime** so the wholesale progress bar updates live on everyone's phone.

## Project Structure

- `App.tsx` — Main mobile screen with pool switcher, live wholesale minimum progress, item filters, and sticky personal cart bar.
- `src/types/groupBuy.ts` — TypeScript domain models for pools, wholesale items, case splits, and settlement ledgers.
- `src/utils/calculations.ts` — Core math engine for wholesale minimum progress, free delivery fee waivers, case-split matching, and per-neighbor cost breakdowns.
- `src/components/PoolProgressCard.tsx` — Visual tracker for the supplier order minimum, free delivery threshold, per-person delivery split, and total group savings.
- `src/components/CatalogItemCard.tsx` — Item card supporting both direct wholesale units and visual slot bars for bulk case splits.
- `src/components/SplitLedgerView.tsx` — Host pickup logistics and itemized per-neighbor cost split & payment status tracker.
- `src/components/ActionModals.tsx` — Modals for pitching new items/case splits to a pool and hosting a new neighborhood group buy.
- `src/services/repository.ts` — Storage adapter supporting local `AsyncStorage` out-of-the-box and optional Supabase cloud sync.
- `supabase/schema.sql` — Ready-to-run PostgreSQL schema with Realtime publications enabled.

  const handleUpdateQuantity = async (itemId: string, delta: number) => {
    if (activePool.status === 'locked') return;

    const targetItem = activePool.items.find((i) => i.id === itemId);

          delta

        return { ...item, claims: updatedClaims };

        participants: updatedParticipants,
        items: updatedItems,

        nextPoolSnapshot,

        activeUser,

  const handleTrimPartialCase = async (itemId: string) => {
    let trimmedClaimsForSync: ParticipantShare[] = [];

    const updatedPools: GroupBuyPool[] = pools.map((pool) => {
      if (pool.id !== activePool.id) return pool;

      const updatedItems = pool.items.map((item) => {
        if (item.id !== itemId) return item;
        const status = getCaseSplitStatus(item);
        if (status.remainderUnits === 0) return item;

        let toRemove = status.remainderUnits;
        const reversed = [...item.claims].reverse();
        const adjustedReversed: ParticipantShare[] = [];

        for (const claim of reversed) {
          if (toRemove <= 0) {
            adjustedReversed.push(claim);
          } else if (claim.quantity <= toRemove) {
            toRemove -= claim.quantity;
          } else {
            adjustedReversed.push({
              ...claim,
              quantity: claim.quantity - toRemove,
            });
            toRemove = 0;
          }
        }

        trimmedClaimsForSync = adjustedReversed.reverse();
        return { ...item, claims: trimmedClaimsForSync };
      });

      return { ...pool, items: updatedItems };
    });

    setPools(updatedPools);
    await PoolRepository.saveLocalPools(updatedPools);

    if (PoolRepository.getCurrentConfig().enabled) {
      await PoolRepository.syncItemClaimsList(
        activePool.id,
        itemId,
        trimmedClaimsForSync
      );
    }

  /**
   * Evaluates pool & case-split threshold transitions after an atomic claim update
   * and returns any newly triggered alerts (e.g., Case Locked, Minimum Met, <=$25 to Free Delivery, Free Delivery Unlocked).
   */
  evaluateClaimMilestones(
    prevPool: GroupBuyPool,
    nextPool: GroupBuyPool,
    item: CatalogItem,
    claimResult: AtomicClaimResult,
    prefs: NotificationPreferences
  ): PoolNotificationAlert[] {
    const newAlerts: PoolNotificationAlert[] = [];
    const prevSummary = getPoolProgressSummary(prevPool);
    const nextSummary = getPoolProgressSummary(nextPool);

    // 1. Bulk Case Completion or Rollover
    if (prefs.caseSplitCompletion) {
      if (claimResult.cappedAtMaxCases) {
        newAlerts.push(
          this.createAlert(
            nextPool,
            'case_locked',
            `🛑 Max Case Cap Reached: ${item.emoji} ${item.name}`,
            `Host capped this bulk item at ${item.maxCases} full case(s) (${claimResult.totalClaimedUnits} units total).`
          )
        );
      } else if (claimResult.justCompletedCase) {
        newAlerts.push(
          this.createAlert(
            nextPool,
            'case_locked',
            `📦 Case #${claimResult.fullCasesCount} Locked: ${item.emoji} ${item.name}!`,
            `All ${item.caseSize}/${item.caseSize} shares claimed! Full wholesale case is now locked into the supplier order.`
          )
        );
      } else if (claimResult.justOpenedNewCase) {
        newAlerts.push(
          this.createAlert(
            nextPool,
            'case_locked',
            `🆕 Case #${claimResult.fullCasesCount + 1} Opened: ${item.emoji} ${item.name}`,
            `Case #${claimResult.fullCasesCount} is full! Started Case #${claimResult.fullCasesCount + 1} (1/${item.caseSize} claimed).`
          )
        );
      }
    }

    // 2. Wholesale Minimum Order Met
    if (!prevSummary.isMinimumMet && nextSummary.isMinimumMet) {
      newAlerts.push(
        this.createAlert(
          nextPool,
          'minimum_met',
          `🎯 Wholesale Minimum Met ($${nextPool.minimumOrderAmount}+)!`,
          `${nextPool.title} reached $${nextSummary.combinedPoolTotal.toFixed(2)}! This week's wholesale delivery is officially a go.`
        )
      );
    }

    // 3. Free Delivery Unlocked or Within $30 Proximity
    if (prefs.freeDeliveryProximity) {
      if (
        !prevSummary.isFreeDeliveryUnlocked &&
        nextSummary.isFreeDeliveryUnlocked
      ) {
        newAlerts.push(
          this.createAlert(
            nextPool,
            'free_delivery_unlocked',
            `🎉 FREE Wholesale Delivery Unlocked!`,
            `${nextPool.title} passed $${nextPool.freeDeliveryThreshold}! The $${nextPool.deliveryFee} supplier delivery fee is now $0.00 for all neighbors.`
          )

        prevSummary.amountNeededForFreeDelivery > 30 &&
        nextSummary.amountNeededForFreeDelivery <= 30 &&
        nextSummary.amountNeededForFreeDelivery > 0

          )
        );
      }
    }

    return newAlerts;
  },

/**
 * Local in-memory / AsyncStorage equivalent of the Postgres `claim_item_share` RPC:
 * Enforces `maxCases` cap, computes whether a case just locked (`justCompletedCase`)
 * or rolled over into a new case (`justOpenedNewCase`).
 */
export function evaluateAtomicClaimTransition(
  item: CatalogItem,
  user: Participant,
  delta: number
): {
  updatedClaims: ParticipantShare[];
  result: AtomicClaimResult;
} {
  const existingUserClaim = item.claims.find((c) => c.userId === user.id);
  const currentUserQty = existingUserClaim ? existingUserClaim.quantity : 0;
  const otherUsersQty = item.claims
    .filter((c) => c.userId !== user.id)
    .reduce((sum, c) => sum + c.quantity, 0);

  const prevTotal = currentUserQty + otherUsersQty;
  let nextUserQty = Math.max(0, currentUserQty + delta);
  let nextTotal = otherUsersQty + nextUserQty;
  let cappedAtMaxCases = false;

  const caseSize = item.mode === 'case_split' && item.caseSize ? item.caseSize : 1;

  if (
    item.mode === 'case_split' &&
    item.maxCases &&
    item.maxCases > 0
  ) {
    const maxAllowedUnits = item.maxCases * caseSize;
    if (nextTotal > maxAllowedUnits) {
      nextUserQty = Math.max(0, maxAllowedUnits - otherUsersQty);
      nextTotal = otherUsersQty + nextUserQty;
      cappedAtMaxCases = true;
    }
  }

  let updatedClaims = [...item.claims];
  if (nextUserQty <= 0) {
    updatedClaims = updatedClaims.filter((c) => c.userId !== user.id);
  } else if (existingUserClaim) {
    updatedClaims = updatedClaims.map((c) =>
      c.userId === user.id ? { ...c, quantity: nextUserQty } : c
    );
  } else {
    updatedClaims.push({
      userId: user.id,
      userName: user.name,
      avatarColor: user.avatarColor,
      quantity: nextUserQty,
    });
  }

  const prevFullCases =
    item.mode === 'case_split' && caseSize > 1
      ? Math.floor(prevTotal / caseSize)
      : prevTotal;
  const nextFullCases =
    item.mode === 'case_split' && caseSize > 1
      ? Math.floor(nextTotal / caseSize)
      : nextTotal;
  const remainderUnits =
    item.mode === 'case_split' && caseSize > 1 ? nextTotal % caseSize : 0;

  const justCompletedCase =
    item.mode === 'case_split' &&
    caseSize > 1 &&
    nextFullCases > prevFullCases &&
    remainderUnits === 0;

  const justOpenedNewCase =
    item.mode === 'case_split' &&
    caseSize > 1 &&
    prevTotal > 0 &&
    prevTotal % caseSize === 0 &&
    nextTotal > prevTotal;

  return {
    updatedClaims,
    result: {
      newUserQuantity: nextUserQty,
      totalClaimedUnits: nextTotal,
      fullCasesCount: nextFullCases,
      remainderUnits,
      justCompletedCase,
      justOpenedNewCase,
      cappedAtMaxCases,
    },
  };
}

  async seedSupabasePools(pools: GroupBuyPool[]): Promise<{ error?: string }> {
    const client = getOrCreateSupabaseClient();
    if (!client) return {};

    try {
      const poolRows = pools.map((p) => ({
        id: p.id,
        join_code: p.joinCode,
        title: p.title,
        supplier_name: p.supplierName,
        supplier_badge: p.supplierBadge,
        category: p.category,
        host_name: p.hostName,
        host_venmo_handle: p.hostVenmoHandle ?? null,
        neighborhood: p.neighborhood,
        pickup_address: p.pickupAddress,
        pickup_window: p.pickupWindow,
        cutoff_time: p.cutoffTime,
        minimum_order_amount: p.minimumOrderAmount,
        delivery_fee: p.deliveryFee,
        free_delivery_threshold: p.freeDeliveryThreshold,
        status: p.status,
      }));

      const participantRows = pools.flatMap((p) =>
        p.participants.map((part) => ({
          pool_id: p.id,
          user_id: part.id,
          user_name: part.name,
          neighborhood: part.neighborhood,
          avatar_color: part.avatarColor,
          is_host: Boolean(part.isHost),
          paid_status: part.paidStatus,
          picked_up: Boolean(part.pickedUp),
        }))
      );

      const itemRows = pools.flatMap((p) =>
        p.items.map((item) => ({
          id: item.id,
          pool_id: p.id,
          name: item.name,
          category: item.category,
          emoji: item.emoji,
          mode: item.mode,
          unit_label: item.unitLabel,
          unit_price: item.unitPrice,
          retail_price: item.retailPrice,
          case_size: item.caseSize ?? null,
          max_cases: item.maxCases ?? null,
        }))
      );

      const claimRows = pools.flatMap((p) =>
        p.items.flatMap((item) =>
          item.claims.map((c) => ({
            pool_id: p.id,
            item_id: item.id,
            user_id: c.userId,
            user_name: c.userName,
            avatar_color: c.avatarColor,
            quantity: c.quantity,
          }))
        )
      );

      const messageRows = pools.flatMap((p) =>
        (p.messages || []).map((m) => ({
          id: m.id,
          pool_id: p.id,
          user_id: m.userId,
          user_name: m.userName,
          avatar_color: m.avatarColor,
          is_host: Boolean(m.isHost),
          is_announcement: Boolean(m.isAnnouncement),
          text: m.text,
          created_at_label: m.createdAt,
        }))
      );

      const r1 = await client.from('pools').upsert(poolRows, { onConflict: 'id' });
      if (r1.error) throw r1.error;

      const r2 = await client
        .from('pool_participants')
        .upsert(participantRows, { onConflict: 'pool_id,user_id' });
      if (r2.error) throw r2.error;

      const r3 = await client.from('catalog_items').upsert(itemRows, { onConflict: 'id' });
      if (r3.error) throw r3.error;

      if (claimRows.length > 0) {
        const r4 = await client
          .from('item_claims')
          .upsert(claimRows, { onConflict: 'item_id,user_id' });
        if (r4.error) throw r4.error;
      }

      if (messageRows.length > 0) {
        await client.from('pool_messages').upsert(messageRows, { onConflict: 'id' });
      }

      return {};
    } catch (err: any) {
      return { error: err?.message || 'Failed to seed Supabase database' };
    }
  },

  /**
   * Calls the PostgreSQL `claim_item_share` RPC (`FOR UPDATE` row-level lock) when
   * connected to Supabase, falling back to direct table upsert if the user hasn't run
   * the updated SQL function yet.
   */
  async syncAtomicClaimDelta(
    poolId: string,
    itemId: string,
    user: Participant,
    delta: number,
    fallbackQuantity: number
  ): Promise<{ rpcResult?: AtomicClaimResult; error?: string }> {
    const client = getOrCreateSupabaseClient();
    if (!client) return {};

    try {
      const { data, error } = await client.rpc('claim_item_share', {
        p_pool_id: poolId,
        p_item_id: itemId,
        p_user_id: user.id,
        p_user_name: user.name,
        p_avatar_color: user.avatarColor,
        p_neighborhood: user.neighborhood,
        p_delta: delta,
      });

      if (!error && data) {
        return { rpcResult: data as AtomicClaimResult };
      }

      // Fallback to standard upsert if RPC isn't created yet
      return await this.syncClaimQuantity(poolId, itemId, user, fallbackQuantity);
    } catch (err: any) {
      return { error: err?.message || 'Atomic claim RPC failed' };
    }
  },

  async syncClaimQuantity(
    poolId: string,
    itemId: string,
    user: Participant,
    nextQuantity: number
  ): Promise<{ error?: string }> {
    const client = getOrCreateSupabaseClient();
    if (!client) return {};

    try {
      await client.from('pool_participants').upsert(
        {
          pool_id: poolId,
          user_id: user.id,
          user_name: user.name,
          neighborhood: user.neighborhood,
          avatar_color: user.avatarColor,
          is_host: Boolean(user.isHost),
          paid_status: user.paidStatus,
        },
        { onConflict: 'pool_id,user_id' }
      );

      if (nextQuantity <= 0) {
        const { error } = await client
          .from('item_claims')
          .delete()
          .match({ item_id: itemId, user_id: user.id });
        if (error) throw error;
      } else {
        const { error } = await client.from('item_claims').upsert(
          {
            pool_id: poolId,
            item_id: itemId,
            user_id: user.id,
            user_name: user.name,
            avatar_color: user.avatarColor,
            quantity: nextQuantity,
            updated_at: new Date().toISOString(),
          },
          { onConflict: 'item_id,user_id' }
        );
        if (error) throw error;
      }
      return {};
    } catch (err: any) {
      return { error: err?.message || 'Failed to sync item claim' };
    }
  },

    const client = getOrCreateSupabaseClient();
    if (!client) {
      return () => {};
    }

    if (activeChannel) {
      activeChannel.unsubscribe();
      activeChannel = null;
    }

    const channel = client
      .channel('poolcart-live-sync')
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'item_claims' },
        () => onRemoteChange('item_claims')
      )
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'catalog_items' },
        () => onRemoteChange('catalog_items')
      )
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'pool_participants' },
        () => onRemoteChange('pool_participants')
      )
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'pools' },
        () => onRemoteChange('pools')
      )
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'pool_messages' },
        () => onRemoteChange('pool_messages')
      )
      .subscribe();

    activeChannel = channel;

    return () => {
      channel.unsubscribe();
      if (activeChannel === channel) {
        activeChannel = null;
      }
    };
  },

import { CatalogItem, GroupBuyPool, UserCostBreakdown } from '../types/groupBuy';

export interface CaseSplitStatus {
  totalClaimedUnits: number;
  caseSize: number;
  fullCasesCount: number;
  committedUnits: number; // Units that are part of full cases
  remainderUnits: number; // Units waiting to fill the next case
  neededForNextCase: number;
  isFullyMatched: boolean;
}

export interface WholesaleManifestLine {
  itemId: string;
  itemName: string;
  emoji: string;
  mode: 'individual' | 'case_split';
  orderQuantityLabel: string; // e.g. "4 units (12-ct bag)" or "1 full case (6 units)"
  committedUnits: number;
  unfilledRemainderUnits: number;
  neededToFillNextCase: number;
  unitPrice: number;
  lineTotal: number;
}

/**
 * Calculates how many full cases are formed for a `case_split` item,
 * and how many units are still needed to complete the current partial case.
 */
export function getCaseSplitStatus(item: CatalogItem): CaseSplitStatus {
  const totalClaimedUnits = item.claims.reduce((sum, c) => sum + c.quantity, 0);
  if (item.mode !== 'case_split' || !item.caseSize || item.caseSize <= 1) {
    return {
      totalClaimedUnits,
      caseSize: 1,
      fullCasesCount: totalClaimedUnits,
      committedUnits: totalClaimedUnits,
      remainderUnits: 0,
      neededForNextCase: 0,
      isFullyMatched: totalClaimedUnits > 0,
    };
  }

  const caseSize = item.caseSize;
  const fullCasesCount = Math.floor(totalClaimedUnits / caseSize);
  const committedUnits = fullCasesCount * caseSize;
  const remainderUnits = totalClaimedUnits % caseSize;
  const neededForNextCase = remainderUnits === 0 ? 0 : caseSize - remainderUnits;

  return {
    totalClaimedUnits,
    caseSize,
    fullCasesCount,
    committedUnits,
    remainderUnits,
    neededForNextCase,
    isFullyMatched: totalClaimedUnits > 0 && remainderUnits === 0,
  };
}

/**
 * Calculates pool-wide wholesale progress:
 * - committedPoolTotal: subtotal of all individual items + completed cases
 * - pendingCaseTotal: subtotal of partial cases waiting for more neighbors
 * - delivery fee status (waived once freeDeliveryThreshold is reached, otherwise split evenly among active buyers)
 */
export function getPoolProgressSummary(pool: GroupBuyPool) {
  let committedPoolTotal = 0;
  let pendingCaseTotal = 0;
  let totalRetailValue = 0;

  pool.items.forEach((item) => {
    const status = getCaseSplitStatus(item);
    committedPoolTotal += status.committedUnits * item.unitPrice;
    pendingCaseTotal += status.remainderUnits * item.unitPrice;
    totalRetailValue += status.totalClaimedUnits * item.retailPrice;
  });

  const combinedPoolTotal = committedPoolTotal + pendingCaseTotal;
  const minimumProgressRatio = Math.min(1, combinedPoolTotal / pool.minimumOrderAmount);
  const freeDeliveryProgressRatio = Math.min(1, combinedPoolTotal / pool.freeDeliveryThreshold);

  const amountNeededForMinimum = Math.max(0, pool.minimumOrderAmount - combinedPoolTotal);
  const amountNeededForFreeDelivery = Math.max(0, pool.freeDeliveryThreshold - combinedPoolTotal);

  const isMinimumMet = combinedPoolTotal >= pool.minimumOrderAmount;
  const isFreeDeliveryUnlocked = combinedPoolTotal >= pool.freeDeliveryThreshold;

  // Count participants who have claimed at least 1 unit across any item
  const activeParticipantIds = new Set<string>();
  pool.items.forEach((item) => {
    item.claims.forEach((c) => {
      if (c.quantity > 0) activeParticipantIds.add(c.userId);
    });
  });

  const activeBuyerCount = Math.max(1, activeParticipantIds.size);
  const currentDeliveryFeePerPerson = isFreeDeliveryUnlocked
    ? 0
    : pool.deliveryFee / activeBuyerCount;

  const totalPoolSavings = Math.max(0, totalRetailValue - combinedPoolTotal);

  return {
    committedPoolTotal,
    pendingCaseTotal,
    combinedPoolTotal,
    minimumProgressRatio,
    freeDeliveryProgressRatio,
    amountNeededForMinimum,
    amountNeededForFreeDelivery,
    isMinimumMet,
    isFreeDeliveryUnlocked,
    activeBuyerCount,
    currentDeliveryFeePerPerson,
    totalPoolSavings,
  };
}

/**
 * Generates per-participant itemized ledger and totals for checkout & host tracking.
 */
export function getParticipantBreakdowns(pool: GroupBuyPool): UserCostBreakdown[] {
  const summary = getPoolProgressSummary(pool);

  return pool.participants.map((participant) => {
    let committedSubtotal = 0;
    let pendingCaseSubtotal = 0;
    let totalEstimatedSavings = 0;
    const lineItems: UserCostBreakdown['lineItems'] = [];

    pool.items.forEach((item) => {
      const claim = item.claims.find((c) => c.userId === participant.id);
      if (!claim || claim.quantity <= 0) return;

      const status = getCaseSplitStatus(item);
      const itemSubtotal = claim.quantity * item.unitPrice;
      const itemSavings = claim.quantity * Math.max(0, item.retailPrice - item.unitPrice);
      totalEstimatedSavings += itemSavings;

      const isCaseSplit = item.mode === 'case_split';
      const caseFilled = !isCaseSplit || status.remainderUnits === 0;

      if (caseFilled) {
        committedSubtotal += itemSubtotal;
      } else {
        pendingCaseSubtotal += itemSubtotal;
      }

      lineItems.push({
        itemId: item.id,
        itemName: item.name,
        emoji: item.emoji,
        quantity: claim.quantity,
        unitPrice: item.unitPrice,
        subtotal: itemSubtotal,
        isCaseSplit,
        caseFilled,
        caseProgressText: isCaseSplit
          ? `${status.totalClaimedUnits}/${status.caseSize} case units claimed`
          : undefined,
      });
    });

    const hasItems = lineItems.length > 0;
    const estimatedDeliveryShare = hasItems ? summary.currentDeliveryFeePerPerson : 0;
    const totalDue = committedSubtotal + pendingCaseSubtotal + estimatedDeliveryShare;

    return {
      userId: participant.id,
      userName: participant.name,
      avatarColor: participant.avatarColor,
      committedSubtotal,
      pendingCaseSubtotal,
      estimatedDeliveryShare,
      totalEstimatedSavings,
      totalDue,
      paidStatus: participant.paidStatus,
      pickedUp: participant.pickedUp,
      lineItems,
    };
  });
}

/**
 * Generates the Host Wholesale Supplier Purchase Order Manifest
 */
export function getWholesaleOrderManifest(pool: GroupBuyPool): {
  lines: WholesaleManifestLine[];
  partialCaseItems: WholesaleManifestLine[];
  totalSupplierMerchandiseCost: number;
  supplierDeliveryFee: number;
  grandTotalSupplierInvoice: number;
} {
  const summary = getPoolProgressSummary(pool);
  const lines: WholesaleManifestLine[] = [];
  const partialCaseItems: WholesaleManifestLine[] = [];

  pool.items.forEach((item) => {
    const status = getCaseSplitStatus(item);
    if (status.totalClaimedUnits === 0) return;

    const lineTotal = status.committedUnits * item.unitPrice;
    const orderQuantityLabel =
      item.mode === 'case_split'
        ? `${status.fullCasesCount} full case${status.fullCasesCount === 1 ? '' : 's'} (${status.committedUnits} × ${item.unitLabel})`
        : `${status.committedUnits} × ${item.unitLabel}`;

    const entry: WholesaleManifestLine = {
      itemId: item.id,
      itemName: item.name,
      emoji: item.emoji,
      mode: item.mode,
      orderQuantityLabel,
      committedUnits: status.committedUnits,
      unfilledRemainderUnits: status.remainderUnits,
      neededToFillNextCase: status.neededForNextCase,
      unitPrice: item.unitPrice,
      lineTotal,
    };

    lines.push(entry);
    if (item.mode === 'case_split' && status.remainderUnits > 0) {
      partialCaseItems.push(entry);
    }
  });

  const totalSupplierMerchandiseCost = lines.reduce((sum, l) => sum + l.lineTotal, 0);
  const supplierDeliveryFee = summary.isFreeDeliveryUnlocked ? 0 : pool.deliveryFee;
  const grandTotalSupplierInvoice = totalSupplierMerchandiseCost + supplierDeliveryFee;

  return {
    lines,
    partialCaseItems,
    totalSupplierMerchandiseCost,
    supplierDeliveryFee,
    grandTotalSupplierInvoice,
  };
}

/**
 * Generates formatted text for native iOS/Android sharing (invite neighbors or export supplier manifest)
 */
export function buildShareablePoolInviteText(pool: GroupBuyPool): string {
  const summary = getPoolProgressSummary(pool);
  const partialCases = pool.items
    .map((item) => ({ item, status: getCaseSplitStatus(item) }))

  const partialLines =

        partialCases
          .map(
            (x) =>
              `• ${x.item.emoji} ${x.item.name}: ${x.status.neededForNextCase} more needed ($${x.item.unitPrice.toFixed(2)}/${x.item.unitLabel})`
          )
          .join('\n')
      : '';

  return (
    `🥬 Join our neighborhood grocery group buy on PoolCart!\n` +
    `Pool: ${pool.title} (${pool.supplierName})\n` +
    `Host: ${pool.hostName} • Pickup: ${pool.pickupWindow} at ${pool.pickupAddress}\n` +
    `Progress: $${summary.combinedPoolTotal.toFixed(2)} / $${pool.minimumOrderAmount} minimum (${
      summary.isFreeDeliveryUnlocked

export function buildSupplierOrderExportText(pool: GroupBuyPool): string {
  const manifest = getWholesaleOrderManifest(pool);
  const breakdowns = getParticipantBreakdowns(pool).filter((b) => b.lineItems.length > 0);

  const itemLines = manifest.lines
    .filter((l) => l.committedUnits > 0)
    .map(
      (l) =>
        `• ${l.emoji} ${l.itemName}: ${l.orderQuantityLabel} — $${l.lineTotal.toFixed(2)}`
    )
    .join('\n');

  const neighborLines = breakdowns
    .map(
      (b) =>
        `• ${b.userName} (${b.paidStatus === 'paid' ? 'PAID' : 'PENDING'}): $${b.totalDue.toFixed(2)} — ` +
        b.lineItems.map((li) => `${li.emoji}×${li.quantity}`).join(', ')
    )
    .join('\n');

  return (
    `📋 WHOLESALE PURCHASE ORDER — ${pool.title}\n` +
    `Supplier: ${pool.supplierName}\n` +
    `Host: ${pool.hostName} | Pickup: ${pool.pickupWindow}\n\n` +
    `--- SUPPLIER MASTER ORDER ---\n` +
    `${itemLines || '(No committed full units yet)'}\n` +
    `Merchandise Subtotal: $${manifest.totalSupplierMerchandiseCost.toFixed(2)}\n` +
    `Wholesale Delivery Fee: $${manifest.supplierDeliveryFee.toFixed(2)}\n` +
    `Total Supplier Invoice: $${manifest.grandTotalSupplierInvoice.toFixed(2)}\n\n` +
    `--- NEIGHBOR PICKUP & SPLIT CHECKLIST ---\n` +
    `${neighborLines}`
  );

  const myQty = userClaim ? userClaim.quantity : 0;

    ((item.retailPrice - item.unitPrice) / item.retailPrice) * 100
  );

  const isMaxCasesCapped =
    item.mode === 'case_split' &&
    Boolean(
      item.maxCases &&
        item.caseSize &&
        caseStatus.totalClaimedUnits >= item.maxCases * item.caseSize
    );

  // Build slot indicators for case_split mode
  const renderCaseSlots = () => {
    if (item.mode !== 'case_split' || !item.caseSize) return null;

    const slotsInCurrentView =
      caseStatus.remainderUnits === 0 && caseStatus.totalClaimedUnits > 0
        ? item.caseSize
        : caseStatus.remainderUnits;

    const activeCaseNumber =
      caseStatus.remainderUnits === 0 && caseStatus.fullCasesCount > 0
        ? caseStatus.fullCasesCount
        : caseStatus.fullCasesCount + 1;

    const slots = Array.from({ length: item.caseSize }, (_, idx) => {
      const filled = idx < slotsInCurrentView;
      return (
        <View
          key={idx}
          style={[
            styles.caseSlot,
            filled ? styles.caseSlotFilled : styles.caseSlotEmpty,
          ]}
        />
      );
    });

    return (
      <View style={styles.caseSplitContainer}>
        <View style={styles.caseHeaderRow}>
          <Text style={styles.caseTitle}>
            📦 Case #{activeCaseNumber} ({item.caseSize} units/case
            {item.maxCases ? ` • Max ${item.maxCases} cases` : ''})
          </Text>
          <Text
            style={[
              styles.caseStatusBadge,
              caseStatus.neededForNextCase === 0 && caseStatus.totalClaimedUnits > 0
                ? styles.caseMatchedText
                : styles.casePendingText,
            ]}
          >
            {isMaxCasesCapped
              ? `🔒 All ${item.maxCases} case(s) full!`
              : caseStatus.totalClaimedUnits === 0
              ? `0/${item.caseSize} claimed`
              : caseStatus.neededForNextCase === 0
              ? `✓ ${caseStatus.fullCasesCount} full case(s) locked! (+ opens Case #${
                  caseStatus.fullCasesCount + 1
                })`
              : `${caseStatus.neededForNextCase} more needed for Case #${activeCaseNumber}`}
          </Text>
        </View>
        <View style={styles.slotsRow}>{slots}</View>
      </View>
    );
  };

  return (

}) => {

  const pinnedAnnouncements = messages.filter((m) => m.isAnnouncement);

    .map((item) => ({ item, status: getCaseSplitStatus(item) }))

      (x) => x.item.mode === 'case_split' && x.status.neededForNextCase > 0
    );

    if (!draft.trim()) return;

    setDraft('');
    setPostAsAnnouncement(false);
  };

              </Text>
            </View>
          ))}

import React from 'react';
import {
  View,
  Text,
  StyleSheet,
  TouchableOpacity,
  Linking,
  Alert,
} from 'react-native';
import { GroupBuyPool } from '../types/groupBuy';
import { getParticipantBreakdowns } from '../utils/calculations';

interface Props {
  pool: GroupBuyPool;
  currentUserId: string;
  onTogglePaidStatus: (userId: string) => void;
}

export const SplitLedgerView: React.FC<Props> = ({
  pool,
  currentUserId,
  onTogglePaidStatus,
}) => {
  const breakdowns = getParticipantBreakdowns(pool);
  const venmoHandle = pool.hostVenmoHandle || 'PoolCart-Host';

  const handlePayWithVenmo = async (
    userId: string,
    amount: number,
    userName: string
  ) => {
    const note = encodeURIComponent(
      `PoolCart: ${pool.title} (${userName.split(' ')[0]})`
    );
    const amountStr = amount.toFixed(2);
    const venmoDeepLink = `venmo://paycharge?txn=pay&recipients=${encodeURIComponent(
      venmoHandle
    )}&amount=${amountStr}&note=${note}`;
    const venmoWebFallback = `https://venmo.com/${encodeURIComponent(
      venmoHandle
    )}?txn=pay&amount=${amountStr}&note=${note}`;

    try {
      const supported = await Linking.canOpenURL(venmoDeepLink);
      if (supported) {
        await Linking.openURL(venmoDeepLink);
      } else {
        await Linking.openURL(venmoWebFallback);
      }
      // Also offer to mark as paid
      if (pool.participants.find((p) => p.id === userId)?.paidStatus !== 'paid') {
        onTogglePaidStatus(userId);
      }
    } catch {
      Alert.alert(
        'Venmo Payment Details',
        `Send $${amountStr} to @${venmoHandle} with note "${pool.title}".`
      );
    }
  };

  return (
    <View style={styles.container}>
      <View style={styles.pickupBanner}>
        <Text style={styles.pickupTitle}>📍 Pickup & Drop-off Logistics</Text>
        <Text style={styles.pickupDetail}>
          Host: {pool.hostName} (@{venmoHandle})
        </Text>
        <Text style={styles.pickupDetail}>Address: {pool.pickupAddress}</Text>
        <Text style={styles.pickupDetail}>Window: {pool.pickupWindow}</Text>
        <Text style={styles.cutoffNote}>
          {pool.status === 'locked'
            ? '🔒 Pool Locked — Ready for Settlement & Pickup'
            : `Order Cutoff: ${pool.cutoffTime}`}
        </Text>
      </View>

      <Text style={styles.sectionHeader}>
        NEIGHBOR COST SPLIT & SETTLEMENT LEDGER
      </Text>

      {breakdowns.map((b) => {
        const isMe = b.userId === currentUserId;
        return (
          <View
            key={b.userId}
            style={[styles.memberCard, isMe && styles.myMemberCard]}
          >
            <View style={styles.memberHeader}>
              <View style={styles.memberNameRow}>
                <View
                  style={[styles.avatar, { backgroundColor: b.avatarColor }]}
                >
                  <Text style={styles.avatarInitial}>
                    {b.userName.charAt(0).toUpperCase()}
                  </Text>
                </View>
                <View>
                  <Text style={styles.memberName}>{b.userName}</Text>
                  <Text style={styles.savingsSub}>
                    Saving ${b.totalEstimatedSavings.toFixed(2)} vs. retail
                  </Text>
                </View>
              </View>

              <TouchableOpacity
                style={[
                  styles.paidChip,
                  b.paidStatus === 'paid'
                    ? styles.paidChipGreen
                    : styles.paidChipAmber,
                ]}
                onPress={() => onTogglePaidStatus(b.userId)}
              >
                <Text
                  style={[
                    styles.paidChipText,
                    b.paidStatus === 'paid'
                      ? styles.paidTextGreen
                      : styles.paidTextAmber,
                  ]}
                >
                  {b.paidStatus === 'paid' ? '✓ Paid' : 'Mark Paid'}
                </Text>
              </TouchableOpacity>
            </View>

            {/* Line Items */}
            {b.lineItems.length === 0 ? (
              <Text style={styles.emptyItemsText}>
                No items claimed in this pool yet.
              </Text>
            ) : (
              <View style={styles.itemsList}>
                {b.lineItems.map((li) => (
                  <View key={li.itemId} style={styles.lineItemRow}>
                    <View style={styles.lineItemLeft}>
                      <Text style={styles.lineItemTitle}>
                        {li.emoji} {li.itemName} (×{li.quantity})
                      </Text>
                      {li.isCaseSplit && (
                        <Text
                          style={[
                            styles.caseSubNote,
                            li.caseFilled
                              ? { color: '#15803D' }
                              : { color: '#9333EA' },
                          ]}
                        >
                          {li.caseFilled
                            ? '✓ Case filled'
                            : `⏳ Waiting for case to fill (${li.caseProgressText})`}
                        </Text>
                      )}
                    </View>
                    <Text style={styles.lineItemPrice}>
                      ${li.subtotal.toFixed(2)}
                    </Text>
                  </View>
                ))}

                <View style={styles.feeRow}>
                  <Text style={styles.feeLabel}>Shared Wholesale Delivery</Text>
                  <Text style={styles.feeValue}>
                    {b.estimatedDeliveryShare === 0
                      ? 'FREE (Waived)'
                      : `$${b.estimatedDeliveryShare.toFixed(2)}`}
                  </Text>
                </View>

                <View style={styles.totalRow}>
                  <Text style={styles.totalLabel}>Total Pool Share</Text>
                  <Text style={styles.totalValue}>
                    ${b.totalDue.toFixed(2)}
                  </Text>
                </View>

                {/* 1-Tap Venmo Deep Link Settlement Button */}
                {b.paidStatus !== 'paid' && b.totalDue > 0 && (
                  <TouchableOpacity
                    style={styles.venmoButton}
                    onPress={() =>
                      handlePayWithVenmo(b.userId, b.totalDue, b.userName)
                    }
                  >
                    <Text style={styles.venmoButtonText}>
                      💸 Pay ${b.totalDue.toFixed(2)} to @{venmoHandle} via Venmo
                    </Text>
                  </TouchableOpacity>
                )}
              </View>
            )}
          </View>
        );
      })}
    </View>
  );
};

-- 6. Atomic Case-Split Claim RPC Function (`claim_item_share`)
-- Uses `FOR UPDATE` row-level locking on `catalog_items` and `pools` so if two
-- neighbors tap the last share of a bulk case at the exact same millisecond:
--   • It serializes their claims cleanly without lost updates
--   • Rejects mutations if the pool was locked by the host
--   • Enforces `max_cases` cap if the host capped the number of bulk cases
--   • Returns exact case completion & rollover metadata to trigger notifications
create or replace function public.claim_item_share(
  p_pool_id text,
  p_item_id text,
  p_user_id text,
  p_user_name text,
  p_avatar_color text,
  p_neighborhood text,
  p_delta integer
)
returns jsonb
language plpgsql
security definer
as $$
declare
  v_pool_status text;
  v_item_mode text;
  v_case_size integer;
  v_max_cases integer;
  v_current_user_qty integer := 0;
  v_other_users_qty integer := 0;
  v_prev_total integer := 0;
  v_next_user_qty integer := 0;
  v_next_total integer := 0;
  v_max_allowed_units integer;
  v_capped boolean := false;
  v_prev_full_cases integer := 0;
  v_next_full_cases integer := 0;
  v_remainder integer := 0;
begin
  -- Lock pool row to verify cutoff lock status
  select status into v_pool_status
  from public.pools
  where id = p_pool_id
  for update;

  if v_pool_status = 'locked' then
    raise exception 'Pool is locked for wholesale ordering; claims cannot be modified.';
  end if;

  -- Lock catalog_item row to serialize concurrent case-split claims
  select mode, coalesce(case_size, 1), max_cases
  into v_item_mode, v_case_size, v_max_cases
  from public.catalog_items
  where id = p_item_id and pool_id = p_pool_id
  for update;

  if not found then
    raise exception 'Catalog item not found in pool.';
  end if;

  -- Ensure participant exists in pool_participants
  insert into public.pool_participants (pool_id, user_id, user_name, neighborhood, avatar_color)
  values (p_pool_id, p_user_id, p_user_name, coalesce(p_neighborhood, 'Victoria Gardens'), p_avatar_color)
  on conflict (pool_id, user_id) do update
  set user_name = excluded.user_name,
      avatar_color = excluded.avatar_color,
      updated_at = now();

  -- Compute current quantities under lock
  select coalesce(quantity, 0) into v_current_user_qty
  from public.item_claims
  where item_id = p_item_id and user_id = p_user_id;

  select coalesce(sum(quantity), 0) into v_other_users_qty
  from public.item_claims
  where item_id = p_item_id and user_id <> p_user_id;

  v_prev_total := v_current_user_qty + v_other_users_qty;
  v_next_user_qty := greatest(0, v_current_user_qty + p_delta);
  v_next_total := v_other_users_qty + v_next_user_qty;

  -- Enforce max_cases cap if set on a case_split item
  if v_item_mode = 'case_split' and v_max_cases is not null and v_max_cases > 0 then
    v_max_allowed_units := v_max_cases * v_case_size;
    if v_next_total > v_max_allowed_units then
      v_next_user_qty := greatest(0, v_max_allowed_units - v_other_users_qty);
      v_next_total := v_other_users_qty + v_next_user_qty;
      v_capped := true;
    end if;
  end if;

  -- Persist updated claim
  if v_next_user_qty <= 0 then
    delete from public.item_claims
    where item_id = p_item_id and user_id = p_user_id;
  else
    insert into public.item_claims (pool_id, item_id, user_id, user_name, avatar_color, quantity, updated_at)
    values (p_pool_id, p_item_id, p_user_id, p_user_name, p_avatar_color, v_next_user_qty, now())
    on conflict (item_id, user_id) do update
    set quantity = excluded.quantity,
        user_name = excluded.user_name,
        avatar_color = excluded.avatar_color,
        updated_at = now();
  end if;

  if v_item_mode = 'case_split' and v_case_size > 1 then
    v_prev_full_cases := floor(v_prev_total::numeric / v_case_size);
    v_next_full_cases := floor(v_next_total::numeric / v_case_size);
    v_remainder := mod(v_next_total, v_case_size);
  else
    v_prev_full_cases := v_prev_total;
    v_next_full_cases := v_next_total;
    v_remainder := 0;
  end if;

  return jsonb_build_object(
    'newUserQuantity', v_next_user_qty,
    'totalClaimedUnits', v_next_total,
    'fullCasesCount', v_next_full_cases,
    'remainderUnits', v_remainder,
    'justCompletedCase', (v_item_mode = 'case_split' and v_next_full_cases > v_prev_full_cases and v_remainder = 0),
    'justOpenedNewCase', (v_item_mode = 'case_split' and v_prev_total > 0 and mod(v_prev_total, v_case_size) = 0 and v_next_total > v_prev_total),
    'cappedAtMaxCases', v_capped
  );
end;
$$;
